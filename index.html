const track=stream.getVideoTracks()[0]; const actual=track.getSettings(); // 取得したカメラが前面と判定されたら使用しない const actualLabel=(track.label||chosen.label||"").toLowerCase(); if(/front|user|前面|インカメラ|true.?depth/.test(actualLabel)){ stream.getTracks().forEach(t=>t.stop()); setStatus("カメラを切り替えられませんでした。\n"+err.message); }finally{ switching=false; $("startBtn").disabled=false; $("applyCameraZoom").disabled=false; $("startBtn").textContent=stream?"カメラ起動済み":"カメラを起動"; } }
